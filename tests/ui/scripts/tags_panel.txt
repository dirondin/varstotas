# Tags card: the eye of Untagged hides untagged top-level geometry (and comes back), the trash
# button deletes a tag (its entities become Untagged, one undo step). Real mouse.
box 0 0 0 2000 1000 500
view iso
zoom_extents
action tag_toggle
action tag_toggle
wait 20
expect tags=["QA"] untagged_hidden=false
screenshot shots/tags_start.bmp
expect drawn_faces=6
# fold Properties (room below), open the Tags card
action click_spot card_toggle:Entity Info
wait 5
action click_spot card_toggle:Tags
wait 10
screenshot shots/tags_open.bmp
# the eye of Untagged hides the (untagged) box and shows it again
action click_spot tag_eye:Untagged
wait 10
expect untagged_hidden=true drawn_faces=0
screenshot shots/tags_untagged_off.bmp
action click_spot tag_eye:Untagged
wait 10
expect untagged_hidden=false drawn_faces=6
# hide QA, then delete it by its trash button (kernel tests cover the entities it tagged)
action tag_toggle
wait 5
expect tags_hidden=1
action click_spot tag_delete
wait 10
expect tags_hidden=0 tags=[] untagged_hidden=false drawn_faces=6
screenshot shots/tags_deleted.bmp
action undo
wait 10
expect tags=["QA"]
