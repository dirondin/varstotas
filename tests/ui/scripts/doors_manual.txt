# Doors made by hand: a carcass of four boards with two boards set over its front, each joined
# to the side next to it with cup hinges from the Joinery window, the boards picked by clicks in
# the view (Shift adds) and the kind from the Kind list with the real mouse.
action script let b = box(vec3(18, 0, 100), 564, 450, 18); b.name = "Bottom"; let l = box(vec3(0, 0, 100), 18, 450, 800); l.name = "Left"; let r = box(vec3(582, 0, 100), 18, 450, 800); r.name = "Right"; let t = box(vec3(18, 0, 882), 564, 450, 18); t.name = "Top"; let dl = box(vec3(0, -18, 100), 299, 18, 800); dl.name = "Door L"; let dr = box(vec3(301, -18, 100), 299, 18, 800); dr.name = "Door R";
action console_close
view iso
zoom_extents
wait 10
rawp click 150 -18 500
wait 5
raw mods shift
rawp click 9 225 900
raw mods none
wait 5
expect I=2
action joinery
wait 10
action click_spot joinery_new_kind
wait 5
action click_spot joinery_new_kind:Hinge
wait 5
action click_spot joinery:add
wait 10
expect Cup hinge Ø35, overlay
raw key Escape
wait 5
rawp click 450 -18 500
wait 5
raw mods shift
rawp click 591 225 900
raw mods none
wait 5
action click_spot joinery:add
wait 10
screenshot shots/doors_manual.bmp
quit
