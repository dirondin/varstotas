# T6c: Procedural Material window (real mouse on its buttons; spots from lib_spots), the new
# material paints a face; physical materials show sphere thumbnails in Materials.
box 0 0 0 1000 1000 1000
view iso
zoom_extents
wait 10
action procedural bricks
wait 5
screenshot shots/proc_bricks.bmp
action click_spot proc:tiles
wait 3
screenshot shots/proc_tiles.bmp
action click_spot proc:planks
wait 3
screenshot shots/proc_planks.bmp
action click_spot proc:create
wait 5
expect mats={"-": 6}
expect paint=Planks
expect tool=Paint
expect undo=Some("Add Material")
rawp click 500 0 500
wait 3
expect mats={"-": 5, "Planks": 1}
# a second one gets a new name
action click_spot proc:noise
wait 3
action click_spot proc:create
wait 5
expect paint=Noise
rawp click 1000 500 500
wait 3
expect mats={"-": 4, "Noise": 1, "Planks": 1}
screenshot shots/proc_painted.bmp
# generator plugins: a tab per plugin, a preview in the background, Create at the full size
action click_spot proc:plugin:basket_weave
wait 40
screenshot shots/proc_basket.bmp
action click_spot proc:create
wait 60
expect paint=Basket Weave
action click_spot proc:plugin:marble
wait 60
screenshot shots/proc_marble.bmp
action procedural plugin:terrazzo
wait 40
screenshot shots/proc_terrazzo.bmp
action click_spot proc:create
wait 60
expect paint=Terrazzo
# sphere thumbnails: a metal and a glass material in the model
action mat_pbr Steel metallic=1 roughness=0.25
action mat_pbr Glass glass=1 roughness=0.02
action tray_only Materials
wait 5
screenshot shots/proc_spheres.bmp
quit
