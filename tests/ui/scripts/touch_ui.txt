# Fingers on the view: one draws (the point goes where it is let go, a loupe above it
# once it is held alone a moment), two pan and pinch, without a click from the first finger.
box 0 0 0 400 300 200
view iso
zoom_extents
wait 10
action ui_compact
wait 10
expect model_faces=6
# a rectangle by two finger taps, each moved before it is let go
tool rectangle
raw finger 0 down 300 760
wait 2
raw finger 0 move 320 770
wait 3
screenshot shots/touch_no_loupe_yet.bmp
# the loupe waits a moment for a second finger
wait 30
screenshot shots/touch_loupe.bmp
raw finger 0 up 320 770
wait 3
# the tapped corner stays shown while the rectangle is drawn
expect tap_marks=1
screenshot shots/touch_tap_mark.bmp
raw finger 0 down 400 820
wait 2
raw finger 0 move 440 860
wait 3
raw finger 0 up 440 860
wait 5
expect model_faces=7
expect tap_marks=0
# two fingers: pinch out zooms in, no click from the first
expect cam_dist=858
raw finger 0 down 800 500
wait 2
raw finger 1 down 1000 500
wait 2
raw finger 0 move 750 500
raw finger 1 move 1050 500
wait 3
expect cam_dist=572 
# moved together: pan
raw finger 0 move 850 550
raw finger 1 move 1150 550
wait 3
expect cam=(462,-326,451)
raw finger 1 up 1150 550
raw finger 0 up 850 550
wait 5
expect model_faces=7
expect tool=Rectangle
