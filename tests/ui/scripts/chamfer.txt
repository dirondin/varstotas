# Chamfer / Fillet: click an edge, move to size it (or type the size), click again;
# typing right after redoes the last one, too big keeps the old result.
box 0 0 0 2000 1000 1000
view iso
zoom_extents
wait 10
tool chamfer
expect tool=Chamfer
# the top front edge (y = 0, z = 1000): click, type the size while sizing
rawp move 1000 0 1000
wait 2
rawp click 1000 0 1000
wait 2
expect model_faces=6
vcb 100
wait 3
expect model_faces=7
expect undo=Some("Chamfer")
# typed right after: the chamfer is redone at 200 mm
vcb 200
wait 3
expect model_faces=7
expect undo=Some("Chamfer")
dump chamfer200
screenshot shots/chamfer.bmp
action undo
wait 2
expect model_faces=6
# click, move away from the edge (live preview), click again
rawp click 1000 0 1000
wait 2
rawp move 1000 0 1150
wait 3
screenshot shots/chamfer_preview.bmp
rawp click 1000 0 1150
wait 3
expect model_faces=7
expect undo=Some("Chamfer")
action undo
wait 2
# Esc cancels sizing
rawp click 1000 0 1000
wait 2
key Escape
wait 2
expect model_faces=6
tool fillet
expect tool=Fillet
vcb 300,8s
wait 2
# double click: the last radius (300)
rawp click 1000 0 1000
wait 1
rawp click 1000 0 1000
wait 3
expect model_faces=14
expect undo=Some("Fillet")
# too big for the 1000 mm faces: the 300 mm fillet stays
vcb 1200
wait 3
expect model_faces=14
expect undo=Some("Fillet")
screenshot shots/fillet.bmp
