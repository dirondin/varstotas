# A fitting placed by hand (real mouse): a handle copied off one door and pasted onto another
# turns out of the face under the cursor, snaps to the face's middle and is mounted on that
# door, drilling it; pasted into the open it is only placed.
action script let a = box(vec3(0, -18, 0), 600, 18, 800); a.name = "Door A"; let b = box(vec3(1000, -18, 0), 600, 18, 800); b.name = "Door B"; let h = fitting(a, "handle"); select(h);
action console_close
wait 5
expect fittings=[("Handle, centres 192", "Door A")]
action copy
view front
zoom_extents
wait 5
action paste
wait 3
rawp move 1290 -18 410
wait 3
screenshot shots/fitting_place_hover.bmp
rawp click 1290 -18 410
wait 5
expect fittings=[("Handle, centres 192", "Door A"), ("Handle, centres 192", "Door B")]
expect fitting_mounts=[([0.5, 0.5], [0, 0]), ([0.5, 0.5], [0, 0])]
expect undo=Some("Paste")
action paste
wait 3
rawp move 800 -18 600
wait 3
rawp click 800 -18 600
wait 5
expect undo=Some("Paste")
expect tool=Select
expect fittings=[("Handle, centres 192", "Door A"), ("Handle, centres 192", "Door B")]
view iso
zoom_extents
wait 10
screenshot shots/fitting_place.bmp
quit
