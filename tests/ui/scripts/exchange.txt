# Exchange formats: export a box group to GLB / 3MF / DXF / DAE, delete it, import it back.
box 0 0 0 2000 1000 500
view iso
zoom_extents
wait 10
select_all
action make_group
wait 2
action export_file models/exchange.glb
action export_file models/exchange.3mf
action export_file models/exchange.dxf
action export_file models/exchange.dae
wait 2
select_all
action delete
wait 2
# (the erased group's definition stays in the model: 6 faces)
expect undo=Some("Erase")
expect sel F=0 E=0 I=0
action import_file models/exchange.glb
wait 5
expect defs=3 model_faces=12
expect undo=Some("Import GLB")
screenshot shots/exchange_glb.bmp
# parts report of the imported group
action export_file models/exchange.csv
action undo
wait 2
expect defs=2 model_faces=6
action import_file models/exchange.dxf
wait 5
# loose faces in the root
expect defs=2 model_faces=12
expect mats={"-": 6}
expect undo=Some("Import DXF")
# COLLADA back: one group again
action undo
wait 2
expect defs=2 model_faces=6
action import_file models/exchange.dae
wait 5
expect defs=3 model_faces=12
expect undo=Some("Import DAE")
screenshot shots/exchange_dae.bmp
# OBJ whose .mtl is not beside it: the geometry loads, the missing file is reported (outside a
# scripted run Varstotas offers to choose the model's folder and imports again from there)
expect import_missing=[]
action import_file ../../tests/ui/fixtures/sidecars/lonely.obj
wait 5
expect import_missing=["lonely.mtl"]
expect undo=Some("Import OBJ")
# OBJ export writes its .mtl beside it
action export_file models/exchange.obj
wait 5
action import_file models/exchange.obj
wait 5
expect import_missing=[]
