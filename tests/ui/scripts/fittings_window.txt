# Window → Fittings (opened from the menu): every fitting on the model by part; the legs opened, their height typed
# (real click and keys) and applied — made again where they were; selected from their line; the
# handle's line deleted; its part's joints opened from a line.
action script let d = box(vec3(0, -18, 0), 600, 18, 800); d.name = "Door"; let t = box(vec3(800, 0, 700), 800, 500, 18); t.name = "Top"; fitting(t, "leg"); fitting(d, "handle"); select([]);
action console_close
wait 5
action click_spot menu:Window
wait 5
action click_spot cmd:Fittings
wait 5
expect fittings_open=true
expect fitting_rows=[(4, "Leg, height 100", "Top"), (1, "Handle, centres 192", "Door")]
action click_spot fittings:row:0
wait 5
action click_spot fittings:row:0:size:0
wait 3
raw mods cmd
raw key A
raw mods none
raw text 300
raw key Enter
wait 5
action click_spot fittings:row:0:style
wait 5
action click_spot fittings:row:0:style:2
wait 5
action click_spot fittings:row:0:apply
wait 5
expect fitting_rows=[(4, "Hairpin leg, height 300", "Top"), (1, "Handle, centres 192", "Door")]
expect hardware=[("Hairpin leg, height 300", 4,
expect undo=Some("Change fittings")
action click_spot fittings:row:0:select
wait 5
expect sel F=0 E=0 I=4
action click_spot fittings:row:1:delete
wait 5
expect fitting_rows=[(4, "Hairpin leg, height 300", "Top")]
expect fittings=[("Hairpin leg, height 300", "Top"), ("Hairpin leg, height 300", "Top"), ("Hairpin leg, height 300", "Top"), ("Hairpin leg, height 300", "Top")]
action click_spot fittings:row:0:joints
wait 5
expect sel F=0 E=0 I=1
view iso
zoom_extents
wait 10
screenshot shots/fittings_window.bmp
quit
