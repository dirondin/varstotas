# Full native workflow: analysis, bed, capped split with sockets/dowels, layout and scoped export.
# Runs in target/qa; qa/run_all.sh writes the stub ./fake-print-slicer there.
box 0 0 0 20 20 20
select_all
action make_group
view iso
zoom_extents
wait 10
action click_spot menu:Window
wait 5
action click_spot cmd:PrintCheck
wait 120
action click_spot printing:tab:Analysis
wait 120
expect print_tools=(Some(Analysis) busy=false intersections=0 complete=true samples=48 unknown=0 thin=0
screenshot shots/printing-analysis.bmp
action print_bed
wait 10
action click_spot printing:center
wait 10
action click_spot printing:bed-refresh
wait 120
expect bed_parts=1 outside=0 overlaps=0
expect stale=false
action click_spot printing:frame-bed
wait 10
screenshot shots/printing-bed.bmp
action print_split
wait 10
action print_cut_set axis=z pos=10 connectors=on
wait 5
screenshot shots/printing-cut-preview.bmp
action click_spot printing:cut
wait 180
expect print_tools=(Some(Split) busy=false
expect sel F=0 E=0 I=4
action print_bed
wait 120
action click_spot printing:arrange
wait 10
action click_spot printing:bed-refresh
wait 180
expect bed_parts=4 outside=0 overlaps=0
expect open=0 nonmanifold=0 winding=0 duplicates=0 inverted=0
screenshot shots/printing-arranged.bmp
action print_map wall
action print_analysis
wait 10
screenshot shots/printing-wall-map.bmp
action print_export
wait 10
action print_export_to models/printing-prepared.3mf
wait 120
expect export=true
action print_slicer ./fake-print-slicer
action click_spot printing:slicer-open
wait 10
screenshot shots/printing-export.bmp
save models/printing-prepared.vrst
undo
wait 10
expect stale=true
action print_open_slicer
wait 10
quit
