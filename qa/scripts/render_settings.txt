# Render window settings: quality presets, image size presets, transparent sky, denoiser,
# depth of field, light and colour grading (live, no new samples).

box 500 500 100 1000 1000 1000
box 2500 500 100 1000 1000 1000
box 4500 500 100 1000 1000 1000
view iso
zoom_extents
wait 10
action mat_pbr Brick roughness=0.4
tool paint
material Brick
rawp click 3000 500 600
rawp click 2500 1000 600
rawp click 3000 1000 1100
tool select
wait 5
action toggle_shadows
action env sky=physical turbidity=3 sun=3
action render_set w=480 h=300 samples=24 bounces=4 denoise=on match=off
# Window → Render… (real mouse)
raw click 408 -106
wait 3
raw click 419 258
wait 5
expect window=true) q=None clamp=20 ai=true dof=As view/f2.8 focus=0 transparent=false grade=neutral
# Quality: Draft (real mouse)
raw click 118 108
wait 3
expect render=(480x300 32spp b3 dn
expect q=Some("Draft") clamp=10
# Preview
raw click 178 108
wait 3
expect q=Some("Preview") clamp=20
action render_set samples=24
action render_start
wait 3
expect job=Some("24sppSome([480, 300]),denoised")
screenshot shots/rset_plain.bmp
# Colour (live: the same samples, a new image): open the section, then grade
raw click 84 402
wait 3
action render_set temp=9000 sat=0.3 contrast=1.3 vignette=0.6 exposure=0.5 tonemap=agx
wait 5
expect grade=0.5ev,Some("agx"),9000K,0,0.3,1.3,0.6
expect job=Some("24sppSome([480, 300]),denoised")
screenshot shots/rset_graded.bmp
# Undo restores the neutral grade
undo
wait 3
expect grade=neutral
# Transparent sky (real mouse on the checkbox)
raw click 278 215
wait 3
expect transparent=true

wait 3
action render_save render_transparent.png
wait 3
screenshot shots/rset_transparent.bmp

# Depth of field: on at f/1.4, focused on the near cube

screenshot shots/rset_dof.bmp
# Built-in denoiser
action render_set ai=off samples=16 dof=off

wait 3
