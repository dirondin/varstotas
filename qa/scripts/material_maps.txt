# Material Editor → Textures: a map set from a folder (suffixes, OpenGL normal over DirectX,
# unknown suffixes skipped), clear by the row button, the large view, a large image with the
# Downscale button, export + load back, the external editor round trip. Real mouse for buttons.
box 0 0 0 1000 1000 1000
view iso
zoom_extents
wait 10
tool paint
material Grey
rawp click 500 0 500
wait 3
expect mats={"-": 5, "Grey": 1}
action tray_only Material Editor
action mat_maps ../../qa/models/mapset
wait 5
expect slots=[("base", "oak_color.png", [16, 16]), ("normal", "oak_normal_gl.png", [16, 16]), ("roughness", "oak_rough.png", [16, 16]), ("ao", "oak_ao.png", [16, 16])]
expect undo=Some("Load Map Set")
tool select
wait 3
screenshot shots/maps_set.bmp
# clear Roughness with its row button
raw click 1347 478
wait 5
expect slots=[("base", "oak_color.png", [16, 16]), ("normal", "oak_normal_gl.png", [16, 16]), ("ao", "oak_ao.png", [16, 16])]
expect undo=Some("Clear Texture")
# the large view (click the base thumbnail)
raw click 1185 393
wait 5
expect tex_view=Some("base")
screenshot shots/maps_view.bmp
# export the normal map and load it back as the roughness map
action mat_slot_export normal|exported_normal.png
action mat_slot roughness|exported_normal.png
wait 3
expect ("roughness", "exported_normal.png", [16, 16])
# a large base image: the tile width is kept, the row offers Downscale
action mat_slot base|../../qa/models/wide_strip.png
wait 5
expect ("base", "wide_strip.png", [2600, 64])
expect textures=["Grey:1000x25"]
raw click 1250 423
wait 5
expect ("base", "wide_strip.png", [2048, 50])
expect undo=Some("Downscale Texture")
screenshot shots/maps_downscaled.bmp
# the external editor round trip: the file is written, saving it reloads the slot
action mat_slot_edit roughness
wait 3
expect ext_edits=["Grey-roughness.png"]
wait 20
action copy_file ../../qa/models/wide_strip.png|home/.varstotas/cache/edit/Grey-roughness.png
wait 90
expect ("roughness", "exported_normal.png", [2600, 64])
expect undo=Some("Reload Texture")
undo
wait 3
expect ("roughness", "exported_normal.png", [16, 16])
quit
