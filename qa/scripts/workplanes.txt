# Work planes (track П batch 3): Guide Plane tool by three points with a real mouse, drawing in
# the air on the active plane (grid nodes on it), plane of a face by double-click, normal by an
# arrow key, origin dragged by its gizmo handle, angles, activate / hide / delete, undo.
box 0 0 0 2000 1000 1000
view iso
zoom_extents
wait 10
tool guide_plane
expect tool=GuidePlane
# three points: the front face's plane (y = 0), red along +X, normal -Y
rawp click 0 0 0
wait 2
rawp click 2000 0 0
wait 2
rawp click 0 0 1000
wait 3
expect undo=Some("Add Guide Plane")
expect planes=["Plane 1* o(0,0,0) n(0.00,-1.00,0.00) a(90,-90,0)"]
dump workplane
screenshot shots/workplanes_wall.bmp
