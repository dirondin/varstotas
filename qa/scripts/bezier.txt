# Draw → Bezier Curve with the real mouse: clicks make corners, press-drag pulls out handles,
# Enter finishes, a click on the first point closes, a double-click finishes, Undo.
# The bar over the view: Handles / Smooth / Freehand, Line (no face) / Area (a face).
wait 10
view top
parallel on
zoom_extents
wait 5
tool bezier
expect tool=Bezier
expect bezier=(Handles, Line)
# open curve: a corner, a smooth point (dragged handle), a corner, Enter → 12 + 12 edges
rawp click 0 0 0
rawp drag 1000 0 0 1500 600 0
rawp click 2000 0 0
wait 3
screenshot shots/bezier_open.bmp
raw key Enter
wait 3
expect undo=Some("Bezier Curve")
expect active_mesh F=0 E=24
# Line: back on the first point → a closed loop, no face (12 + 12 + 1 edges)
rawp click 0 2000 0
rawp drag 1000 3000 0 1000 3700 0
rawp click 0 4000 0
rawp click 0 2000 0
wait 3
expect active_mesh F=0 E=49
# Area: Enter closes the triangle into a face
raw click 310 23
wait 3
expect bezier=(Handles, Area)
rawp click 3000 0 0
rawp click 4000 0 0
rawp click 3500 1000 0
raw key Enter
wait 3
expect active_mesh F=1 E=52
# Line: two corners and a double-click — one straight edge
raw click 263 23
wait 3
rawp click 3000 2000 0
rawp dblclick 4000 2000 0
wait 3
expect active_mesh F=1 E=53
# Backspace takes back a point, Escape drops the curve
rawp click 3000 3000 0
rawp click 4000 3000 0
raw key Backspace
raw key Escape
wait 3
expect active_mesh F=1 E=53
# farther out: room for the next curves
raw scroll 500 400 -300
wait 5
# Smooth: three clicks, Enter → a curve through them (2 spans × 12)
raw click 114 23
wait 3
expect bezier=(Smooth, Line)
rawp click 0 6000 0
rawp click 1000 7000 0
rawp click 2000 6000 0
wait 3
screenshot shots/bezier_smooth.bmp
raw key Enter
wait 3
expect active_mesh F=1 E=77
# Freehand Area: a stroke closes into a face on release
raw click 186 23
raw click 310 23
wait 3
expect bezier=(Freehand, Area)
rawp down 6000 0 0
rawp move 7000 500 0
rawp move 7500 1500 0
rawp move 7000 2500 0
rawp move 6000 3000 0
rawp up 6000 3000 0
wait 3
expect active_mesh F=2 E=92
# Freehand Line: an open stroke, no new face
raw click 263 23
wait 3
rawp down 6000 5000 0
rawp move 7000 5500 0
rawp move 8000 5000 0
rawp move 9000 5500 0
rawp up 9000 5500 0
wait 3
expect active_mesh F=2 E=101
raw mods cmd
raw key Z
raw mods
wait 3
expect active_mesh F=2 E=92
# back to the defaults for the other scripts
raw click 46 23
wait 3
expect bezier=(Handles, Line)
zoom_extents
wait 5
screenshot shots/bezier.bmp
quit
