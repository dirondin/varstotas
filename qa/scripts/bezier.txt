# Curve tools with the real mouse. Bezier Curve (the toolbar button): clicks make corners,
# press-drag pulls out handles; Smooth Curve: clicks only; Freehand: a stroke. Enter finishes an
# open curve (edges only), a click on the first point closes it into a face, a double-click
# finishes, Undo.
wait 10
view top
parallel on
zoom_extents
wait 5
raw click 271 -65
wait 3
expect tool=Bezier
expect status="Click the first point of the curve; press and drag
# open curve: a corner, a smooth point (dragged handle), a corner, Enter → 12 + 12 edges
rawp click 0 0 0
rawp drag 1000 0 0 1500 600 0
rawp click 2000 0 0
wait 3
screenshot shots/bezier_open.bmp
raw key Enter
wait 3
expect undo=Some("Bezier Curve")
expect active_mesh F=0 E=24
# back on the first point → closed, a face (12 + 12 + 1 edges)
rawp click 0 2000 0
rawp drag 1000 3000 0 1000 3700 0
rawp click 0 4000 0
rawp click 0 2000 0
wait 3
expect active_mesh F=1 E=49
# two corners and a double-click: one straight edge
rawp click 3000 0 0
rawp dblclick 4000 0 0
wait 3
expect active_mesh F=1 E=50
# Backspace takes back a point, Escape drops the curve
rawp click 3000 1000 0
rawp click 4000 1000 0
raw key Backspace
raw key Escape
wait 3
expect active_mesh F=1 E=50
# farther out: room for the next curves
raw scroll 500 400 -300
wait 5
# Smooth: three clicks, Enter → an open curve through them (2 spans × 12)
tool smooth curve
wait 3
expect tool=SmoothCurve
expect undo=Some("Bezier Curve")
rawp click 0 6000 0
rawp click 1000 7000 0
rawp click 2000 6000 0
wait 3
screenshot shots/bezier_smooth.bmp
raw key Enter
wait 3
expect undo=Some("Smooth Curve")
expect active_mesh F=1 E=74
# Smooth, closed on the first point: a face (3 spans × 12)
rawp click 3000 5000 0
rawp click 4000 6500 0
rawp click 5000 5000 0
rawp click 3000 5000 0
wait 3
expect active_mesh F=2 E=110
# Freehand: a stroke that ends at its start closes into a face (6 spans × 3)
tool freehand
wait 3
expect tool=Freehand
rawp down 6000 0 0
rawp move 7000 500 0
rawp move 7500 1500 0
rawp move 7000 2500 0
rawp move 6000 3000 0
rawp move 5500 1500 0
rawp move 6000 50 0
rawp up 6000 50 0
wait 3
expect active_mesh F=3 E=128
expect undo=Some("Freehand")
# Freehand: an open stroke stays a line
rawp down 6000 5000 0
rawp move 7000 5500 0
rawp move 8000 5000 0
rawp move 9000 5500 0
rawp up 9000 5500 0
wait 3
expect active_mesh F=3 E=137
raw mods cmd
raw key Z
raw mods
wait 3
expect active_mesh F=3 E=128
zoom_extents
wait 5
screenshot shots/bezier.bmp
quit
