# Tags card: the eye of Untagged hides untagged top-level geometry (and comes back), the trash
# button deletes a tag (its entities become Untagged, one undo step). Real mouse.
box 0 0 0 2000 1000 500
view iso
zoom_extents
action tag_toggle
action tag_toggle
wait 20
expect tags=["QA"] untagged_hidden=false
screenshot shots/tags_start.bmp
expect drawn_faces=6
# open the Tags card
raw click 1137 567
wait 10
screenshot shots/tags_open.bmp
# the eye of Untagged hides the (untagged) box and shows it again
raw click 1157 622
wait 10
expect untagged_hidden=true drawn_faces=0
screenshot shots/tags_untagged_off.bmp
raw click 1157 622
wait 10
expect untagged_hidden=false drawn_faces=6
